const BACKEND_URL = "https://fuliza-backend-1-o8h6.onrender.com"; const DELIVERY_FEE = 3; async function payDelivery() { const button = document.getElementById("payBtn"); const message = document.getElementById("message"); const phone = localStorage.getItem("fulizaaPhoneNumber") || ""; if (!/^254\d{9}$/.test(phone)) { message.textContent = "A valid M-PESA number is required."; return; } const reference = "DEL-" + Date.now().toString(36).toUpperCase(); button.disabled = true; button.textContent = "Sending STK Push..."; message.textContent = "Please wait. Sending the KSh 3 payment prompt to your M-PESA phone..."; try { const response = await fetch( BACKEND_URL + "/api/stk-push", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ phone: phone, amount: DELIVERY_FEE, reference: reference }) } ); const data = await response.json(); if (!response.ok || !data.success) { throw new Error( data.message || data.error || "Unable to send STK Push." ); } const transactionId = data.transactionId; localStorage.setItem( "fulizaaDeliveryTransactionId", transactionId ); button.textContent = "STK Sent"; message.textContent = "STK Push sent. Check your M-PESA phone and enter your PIN."; checkDeliveryPayment(transactionId); } catch (error) { console.error( "DELIVERY STK ERROR:", error ); button.disabled = false; button.textContent = "Continue"; message.textContent = error.message || "Unable to start the payment."; } } function checkDeliveryPayment(transactionId) { const button = document.getElementById("payBtn"); const message = document.getElementById("message"); let attempts = 0; const maxAttempts = 100; const checker = setInterval(async () => { attempts++; try { const response = await fetch( BACKEND_URL + "/api/payment-status/" + encodeURIComponent(transactionId), { cache: "no-store" } ); const data = await response.json(); const status = String( data.status || "" ).toUpperCase(); if (status === "COMPLETED") { clearInterval(checker); button.disabled = true; button.textContent = "Payment Successful"; message.textContent = "Payment received successfully."; localStorage.setItem( "fulizaaDeliveryStatus", "COMPLETED" ); if (data.mpesaReceipt) { localStorage.setItem( "fulizaaDeliveryMpesaReceipt", data.mpesaReceipt ); } return; } if (status === "FAILED") { clearInterval(checker); button.disabled = false; button.textContent = "Try Again"; message.textContent = "The payment was not completed."; return; } if (attempts >= maxAttempts) { clearInterval(checker); button.disabled = false; button.textContent = "Check Again"; message.textContent = "Payment confirmation is taking longer than expected."; } } catch (error) { console.error( "DELIVERY STATUS ERROR:", error ); if (attempts >= maxAttempts) { clearInterval(checker); button.disabled = false; button.textContent = "Check Again"; message.textContent = "Unable to confirm the payment right now."; } } }, 3000); }
